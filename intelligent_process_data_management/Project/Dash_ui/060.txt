class DataFetcher:

    def __init__(self, host='localhost', port=8086, user='admin', pwd='12345', db='solutionist'):

        self.client = InfluxDBClient(host=host, port=port, username=user, password=pwd, database=db)

        self.measurements = ['joint_torque', 'robot_position','dht22_sensor_data', 'dynamixel_sensor_data']



    def latest_df(self, limit=10):

        frames = []

        for m in self.measurements:

            pts = list(self.client.query(f"SELECT * FROM {m} ORDER BY time DESC LIMIT {limit}").get_points())

            if pts:

                df = pd.DataFrame(pts[::-1])

                df['time'] = pd.to_datetime(df['time'], utc=True, errors='coerce')

                frames.append(df)

        return pd.concat(frames).sort_values('time') if frames else pd.DataFrame()



class GaugeFactory:

    @staticmethod

    def create(title, value, suffix, axis_range, steps, threshold, unit_font=None):

        bar_color = '#fa0404' if title == 'Pressure' else 'red'



        indicator = go.Indicator(

            mode="gauge+number",

            value=value,

            number={'suffix': suffix, 'font': unit_font or {}},

            title={'text': f"<b>{title}</b>", 'font': {'size': 22}},

            gauge={

                'axis': {'range': axis_range, 'dtick': 10},

                'bar': {'color': bar_color, 'thickness': 0.2},

                'steps': steps,

                'threshold': threshold

            },

            domain={'x': [0,1], 'y': [0,1]}

        )

        fig = go.Figure(indicator)

        fig.update_layout(

            paper_bgcolor="rgba(0,0,0,0)",

            font={'color': 'white', 'family': 'Orbitron'},

            margin=dict(t=0, b=0, l=0, r=0)

        )

        return fig

        def update(n):

            df = self.fetcher.latest_df()

            if df.empty:

                empty = html.Div('데이터 없음', style={'color':'white'})

                return [go.Figure()] * 7 + [empty]



            latest = df.iloc[-1]

            alerts = self.alarm_system.check_values(latest)



            # Create all figures

            pg = self.create_gauge_figure('pressure')

            sg = self.create_gauge_figure('speed')

            hum = self.create_humidity_figure()

            tmp = self.create_temperature_figure()

            pie = self.create_pie_figure()



            # Torque Monitoring

            joints = [f'Joint {i}' for i in range(6)]

            values = [latest.get(f'joint_{i}', 0) for i in range(6)]

            safe = [min(v, 2) for v in values]

            caution = [min(max(v-2, 0), 0.5) for v in values]

            danger = [max(v-2.5, 0) for v in values]



            fig_t = go.Figure()

            fig_t.add_trace(go.Bar(y=joints, x=safe, orientation='h', marker_color='#1f77b4', width=0.3, showlegend=False))

            fig_t.add_trace(go.Bar(y=joints, x=caution, orientation='h', marker_color="#f5e42b", width=0.3, showlegend=False))

            fig_t.add_trace(go.Bar(y=joints, x=danger, orientation='h', marker_color="#eb1e1e", width=0.3, showlegend=False))

            fig_t.update_layout(

                barmode='stack',

                title='로봇 조인트 토크 모니터링',

                yaxis={'autorange':'reversed'},

                xaxis={'range':[0,3], 'title':'Torque (N·m)'},

                **self.common

            )



            annotations = []

            for ji, v in zip(joints, values):

                annotations.append({

                    'x': v, 'y': ji, 'text': f"{v:.2f} N·m",

                    'xanchor': 'left', 'showarrow': False, 'font': {'color': 'white'}

                })

            fig_t.update_layout(annotations=annotations)



            style_dot = {

                'position': 'absolute',

                'top': f"{latest['y']*10}px",

                'left': f"{latest['x']*10}px",

                'width': '10px', 'height': '10px',

                'backgroundColor': 'red', 'borderRadius': '50%'

            }  

            alarm_style = {

                'display': 'block' if alerts else 'none',

                'position': 'fixed',

                'top': '50%',

                'left': '50%',

                'transform': 'translate(-50%, -50%)',

                'backgroundColor': 'rgba(255, 0, 0, 0.9)',

                'padding': '20px',

                'borderRadius': '10px',

                'zIndex': '2000',

                'textAlign': 'center',

                'animation': 'blink 1s infinite'

            }



            return pg, sg, style_dot, hum, dcc.Graph(figure=fig_t, config={'displayModeBar': False}), tmp, pie, alarm_style, '\n'.join(alerts)

       

        """Creates a detailed torque monitoring figure for the modal view"""

        df = self.fetcher.latest_df()

            return go.Figure()

            xaxis={

    def create_gauge_figure(self, gauge_type):

        df = self.fetcher.latest_df()

        if df.empty:

            return go.Figure()

           

        latest = df.iloc[-1]

        if gauge_type == 'pressure':

            return GaugeFactory.create(

                'Pressure', latest['pressure_mpa'], ' Mpa', [0,100],

                [{'range':[0,80],'color':'#1effa3'},

                 {'range':[80,90],'color':'#ffee58'},

                 {'range':[90,100],'color':'#ff5e5e'}],

                {'value':90,'line':{'color':'red','width':4}}

            )

        else:

            return GaugeFactory.create(

                'Injection Speed', latest['linear_velocity_mm_s'], ' mm/s', [0,100],

                [{'range':[0,80],'color':"#1effa3"},

                 {'range':[80,90],'color':'#ffff66'},

                 {'range':[90,100],'color':'#ff6666'}],

                {'value':90,'line':{'color':'red','width':4}}

            )
